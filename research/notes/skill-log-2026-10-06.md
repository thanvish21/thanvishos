### Mini-Lesson
Large Language Model APIs allow applications to generate text, extract structured data, and automate complex workflows over standard HTTP endpoints. Making raw HTTP requests without exponential backoff, timeout configurations, and rate-limit handling causes backend crashes in production environments. Writing resilient wrappers that handle network flakiness, rate limits (HTTP 429), and schema parsing separates production backend engineers from beginners.

### Why Indian Employers Want It
In 2026, Indian tech product companies, GCCs, and high-growth startups are shifting away from hiring pure ML researchers towards Full-Stack and Backend AI Engineers who can reliably wire foundation models into existing production stacks. Demonstrating robust API clients with rate-limiting, error recovery, and zero hardcoded secrets immediately signals junior engineer seniority.

### Working Python Code Example
```python
import json
import os
import time
from urllib.error import HTTPError, URLError
from urllib.request import Request, urlopen


def query_llm_resilient(
    prompt: str,
    model: str = "gpt-4o-mini",
    max_retries: int = 3,
    base_delay: float = 1.0,
) -> str:
    """Makes a resilient HTTP POST request to an OpenAI-compatible LLM endpoint with exponential backoff."""
    api_key = os.environ.get("OPENAI_API_KEY")
    if not api_key:
        raise ValueError("OPENAI_API_KEY environment variable is not set.")

    url = "https://api.openai.com/v1/chat/completions"
    payload = json.dumps(
        {
            "model": model,
            "messages": [{"role": "user", "content": prompt}],
            "temperature": 0.2,
        }
    ).encode("utf-8")

    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {api_key}",
        "User-Agent": "SRM-SkillBuilder/1.0",
    }

    for attempt in range(1, max_retries + 1):
        try:
            req = Request(url, data=payload, headers=headers, method="POST")
            with urlopen(req, timeout=15) as response:
                result = json.loads(response.read().decode("utf-8"))
                return result["choices"][0]["message"]["content"].strip()
        except HTTPError as e:
            if e.code in (429, 500, 502, 503, 504) and attempt < max_retries:
                delay = base_delay * (2 ** (attempt - 1))
                print(
                    f"[Retry {attempt}/{max_retries}] HTTP {e.code}. Retrying in {delay}s..."
                )
                time.sleep(delay)
            else:
                error_body = e.read().decode("utf-8", errors="ignore")
                raise RuntimeError(
                    f"HTTP Error {e.code}: {error_body}"
                ) from None
        except URLError as e:
            if attempt < max_retries:
                delay = base_delay * (2 ** (attempt - 1))
                print(
                    f"[Retry {attempt}/{max_retries}] Network error: {e.reason}. Retrying in {delay}s..."
                )
                time.sleep(delay)
            else:
                raise RuntimeError(f"Connection failed: {e.reason}") from None

    raise RuntimeError("Max retries exceeded without success.")


if __name__ == "__main__":
    # Test call (ensure OPENAI_API_KEY is exported or replace with mock/local server)
    try:
        if "OPENAI_API_KEY" in os.environ:
            output = query_llm_resilient(
                "In one sentence, why is caching crucial in distributed systems?"
            )
            print("Response:", output)
        else:
            print("Set OPENAI_API_KEY to execute live test.")
    except Exception as err:
        print("Execution failed:", err)
```

**Common Mistake:** Failing to set request timeouts or catching generic `except Exception:` without inspecting HTTP status codes, leading to hanging threads or retrying permanent 400 Bad Request / 401 Unauthorized errors forever.
