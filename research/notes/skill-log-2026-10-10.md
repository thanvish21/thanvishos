### Mini-Lesson
Modern backend APIs use asynchronous database drivers to execute high-concurrency SQL queries without blocking Python's event loop. SQLAlchemy 2.0 couples strong typing (`Mapped[...]`, `mapped_column`) with native `asyncio` support via `create_async_engine` and `AsyncSession`. Organizing database operations into the **Repository Pattern** decouples raw SQL persistence from business logic, ensuring codebases remain testable, modular, and maintainable.

In October 2026 hiring data across top Indian product companies, fintech startups, and GCCs, Python was requested in **35.3%** and SQL in **20.1%** of entry-level job descriptions. ### Working Python Code Example
```python
import asyncio
from typing import List, Optional
from sqlalchemy import Integer, String, select
from sqlalchemy.ext.asyncio import (
    AsyncSession,
    async_sessionmaker,
    create_async_engine,
)
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column


# 1. Define Declarative Base & Strongly-Typed Domain Model
class Base(DeclarativeBase):
    pass


class Student(Base):
    __tablename__ = "students"

    id: Mapped[int] = mapped_column(
        Integer, primary_key=True, autoincrement=True
    )
    reg_number: Mapped[str] = mapped_column(String(30), unique=True, index=True)
    full_name: Mapped[str] = mapped_column(String(100), nullable=False)
    cgpa: Mapped[float] = mapped_column(default=0.0)


# 2. Repository Pattern: Encapsulate Async CRUD & Query Logic
class StudentRepository:

    def __init__(self, session: AsyncSession):
        self.session = session

    async def add(self, reg_number: str, full_name: str, cgpa: float) -> Student:
        student = Student(reg_number=reg_number, full_name=full_name, cgpa=cgpa)
        self.session.add(student)
        await self.session.commit()
        await self.session.refresh(student)
        return student

    async def get_by_reg_number(self, reg_number: str) -> Optional[Student]:
        query = select(Student).where(Student.reg_number == reg_number)
        result = await self.session.execute(query)
        return result.scalar_one_or_none()

    async def get_top_performers(self, min_cgpa: float = 9.0) -> List[Student]:
        query = (
            select(Student)
            .where(Student.cgpa >= min_cgpa)
            .order_by(Student.cgpa.desc())
        )
        result = await self.session.execute(query)
        return list(result.scalars().all())


# 3. Execution Pipeline with In-Memory Async SQLite
async def main():
    # Requires: pip install sqlalchemy aiosqlite
    engine = create_async_engine("sqlite+aiosqlite:///:memory:", echo=False)
    session_factory = async_sessionmaker(engine, expire_on_commit=False)

    # Initialize schema asynchronously
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)

    # Perform repository operations
    async with session_factory() as session:
        repo = StudentRepository(session)

        # Insert records
        await repo.add("RA2611027010109", "Thanvish Reddy", 9.6)
        await repo.add("RA2611027010110", "Aditya Sharma", 8.8)
        await repo.add("RA2611027010111", "Meera Nair", 9.4)

        # Query single record
        student = await repo.get_by_reg_number("RA2611027010109")
        if student:
            print(f"Found Student: {student.full_name} | CGPA: {student.cgpa}")

        # Query filtered list
        top_students = await repo.get_top_performers(min_cgpa=9.0)
        print(f"Top Performers count: {len(top_students)}")
        for s in top_students:
            print(f" - {s.reg_number}: {s.full_name} ({s.cgpa})")

    await engine.dispose()


if __name__ == "__main__":
    asyncio.run(main())
```

**Common Mistake:** Using standard synchronous drivers (like `sqlite3` or `psycopg2`) inside an `async def` route/handler, or forgetting `expire_on_commit=False` in `async_sessionmaker`, which causes lazy-loading attribute errors (`DetachedInstanceError`) after the session context exits.
